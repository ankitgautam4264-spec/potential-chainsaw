# Gem dhoondhne wala function - Trending coins
def get_trending_gems():
    try:
        url = "https://api.coingecko.com/api/v3/search/trending"
        r = requests.get(url, timeout=15).json()
        msg = "💎 *Bhoot Ka Khajana - Top Gems* 💎\n\n"
        for i, coin in enumerate(r['coins'][:7], 1):
            item = coin['item']
            name = item['name']
            symbol = item['symbol']
            msg += f"{i}. {name} ({symbol})\n"
        msg += "\n⚠️ Lala ye sirf trending hai, khud research karna!"
        return msg
    except Exception as e:
        return f"Error lala: {e}"

# /start command
async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    chat_id = update.effective_chat.id
    await update.message.reply_text(
        f"👻 *Bhoot Ka Khajana ON Hai Lala!* 👻\n\n"
        f"Tera Chat ID: `{chat_id}`\n"
        f"Isko copy kar le lala, Render me dalna hai!\n\n"
        f"Commands:\n"
        f"/findgems - Abhi ke gems\n"
        f"/getid - Tera ID\n",
        parse_mode="Markdown"
    )

# /getid command
async def getid(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text(f"{update.effective_chat.id}")

# /findgems command
async def findgems(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await update.message.reply_text("Khajana dhoondh raha hu lala... ⏳")
    gems = get_trending_gems()
    await update.message.reply_text(gems, parse_mode="Markdown")

# HAR GHANTE AUTO ALERT - YEHI TO CHAHIYE THA LALA
async def hourly_alert(context: ContextTypes.DEFAULT_TYPE):
    if not MY_CHAT_ID:
        return
    gems = get_trending_gems()
    try:
        await context.bot.send_message(chat_id=MY_CHAT_ID, text=f"🚨 *AUTO ALERT LALA* 🚨\n\n{gems}", parse_mode="Markdown")
        print("Alert bhej diya lala")
    except Exception as e:
        print(f"Fail: {e}")

def main():
    app = Application.builder().token(BOT_TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(CommandHandler("getid", getid))
    app.add_handler(CommandHandler("findgems", findgems))
    
    # Har 3600 second = 1 ghanta me alert
    app.job_queue.run_repeating(hourly_alert, interval=3600, first=10)
    
    print("BhootKaKhajana Bot Running Hai Lala...")
    app.run_polling()

if __name__ == "__main__":
    main()
